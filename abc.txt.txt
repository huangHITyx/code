#include <iostream>

int main()
{
    // 直接打印数字
    std::cout << 10 << std::endl;
    std::cout << 3.1415 << std::endl;
    std::cout << 999 << std::endl;

    // 先把数字存到变量里，再打印变量
    int a = 2026;
    double b = 123.456;
    std::cout << a << std::endl;
    std::cout << b << std::endl;

    return 0;
}
